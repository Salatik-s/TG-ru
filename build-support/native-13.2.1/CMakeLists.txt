cmake_minimum_required(VERSION 3.22)
project(familytgnet LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "familytgnet targets Android NDK only")
endif()
if(NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
  message(FATAL_ERROR "This release pins crypto for arm64-v8a and x86_64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FAMILYGRAM_UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/upstream" CACHE PATH "Verified pinned upstream inputs")
set(FAMILYGRAM_GENERATED "${CMAKE_CURRENT_SOURCE_DIR}/generated" CACHE PATH "prepare.mjs output")
set(TGNET_JNI "${FAMILYGRAM_UPSTREAM}/TMessagesProj/jni")
if(NOT EXISTS "${FAMILYGRAM_GENERATED}/provenance.json")
  message(FATAL_ERROR "Run node prepare.mjs before CMake")
endif()
add_library(familycrypto STATIC IMPORTED)
set_target_properties(familycrypto PROPERTIES IMPORTED_LOCATION "${TGNET_JNI}/boringssl/lib/${ANDROID_ABI}/libcrypto.a")
set(TGNET_FILES ApiScheme BuffersStorage ByteArray ByteStream Connection ConnectionSession ConnectionsManager Datacenter EventObject FileLog MTProtoScheme NativeByteBuffer Request Timer TLObject ProxyCheckInfo Handshake Config)
set(TGNET_SOURCES)
foreach(name IN LISTS TGNET_FILES)
  list(APPEND TGNET_SOURCES "${TGNET_JNI}/tgnet/${name}.cpp")
endforeach()
add_library(familytgnet SHARED ${TGNET_SOURCES}
  "${TGNET_JNI}/TgNetWrapper.cpp"
  "${TGNET_JNI}/boringssl/ssl/aes_ige.c"
  "${FAMILYGRAM_GENERATED}/ConnectionSocket.cpp"
  FamilyTransport.cpp FamilyJni.cpp)
target_include_directories(familytgnet PRIVATE "${TGNET_JNI}" "${TGNET_JNI}/tgnet" "${TGNET_JNI}/boringssl/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(familytgnet PRIVATE ANDROID HAVE_PTHREAD)
target_compile_options(familytgnet PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-write-strings -fvisibility=hidden -fvisibility-inlines-hidden -ffunction-sections -fdata-sections -fno-strict-aliasing -fstack-protector-strong -Oz)
target_link_options(familytgnet PRIVATE -Wl,--no-undefined -Wl,-Bsymbolic -Wl,--exclude-libs,ALL -Wl,--gc-sections "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" -Wl,-z,relro -Wl,-z,now -Wl,-z,max-page-size=16384)
target_link_libraries(familytgnet PRIVATE familycrypto android log z dl m)
set_target_properties(familytgnet PROPERTIES LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
