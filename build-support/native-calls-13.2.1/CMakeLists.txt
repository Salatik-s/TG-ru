cmake_minimum_required(VERSION 3.22)
project(familycalls LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(XHOOK ${CMAKE_CURRENT_SOURCE_DIR}/third_party/xhook/libxhook/jni)
add_library(familycalls SHARED adapter.cc install.cc syscall_gate.S
  ${XHOOK}/xhook.c ${XHOOK}/xh_core.c ${XHOOK}/xh_elf.c ${XHOOK}/xh_log.c ${XHOOK}/xh_util.c ${XHOOK}/xh_version.c)
target_include_directories(familycalls PRIVATE ${XHOOK})
target_compile_options(familycalls PRIVATE -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(familycalls PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384 -Wl,-Bsymbolic "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
target_link_libraries(familycalls PRIVATE log dl)
add_executable(fgcalls-adapter-smoke tests/adapter_smoke.cc adapter.cc syscall_gate.S)
target_link_libraries(fgcalls-adapter-smoke PRIVATE dl)
target_compile_options(fgcalls-adapter-smoke PRIVATE -Wall -Wextra -Werror)
add_executable(fgcalls-audit tests/audit.cc)
target_link_libraries(fgcalls-audit PRIVATE dl)
